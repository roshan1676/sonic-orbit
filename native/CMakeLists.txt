cmake_minimum_required(VERSION 3.20)
project(SonicOrbitDSP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(sonicorbit_dsp SHARED spatial.cpp)
target_compile_definitions(sonicorbit_dsp PRIVATE SO_EXPORTS)
if(MSVC)
  target_compile_options(sonicorbit_dsp PRIVATE /O2 /W4)
endif()
add_executable(sonicorbit_dsp_tests tests.cpp)
target_link_libraries(sonicorbit_dsp_tests PRIVATE sonicorbit_dsp)
enable_testing()
add_test(NAME spatial_core COMMAND sonicorbit_dsp_tests)
